cmake_minimum_required(VERSION 3.20)
project(AuditionMacro VERSION 0.11.6 LANGUAGES C CXX)
foreach(source analysis.cpp zwicker.cpp lufs_meter.cpp)
  if(NOT EXISTS "${CMAKE_SOURCE_DIR}/native/${source}")
    message(FATAL_ERROR "The analysis engine source is not in the public checkout. Use the prebuilt release helper; see docs/source-distribution.md.")
  endif()
endforeach()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
option(AM_TEST_LUA "Download pinned Lua and run shared Lua regressions" OFF)
if(NOT WIN32)
  message(FATAL_ERROR "Use sh native/build.sh on macOS")
endif()
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "REAPER's Windows C++ audio ABI requires MSVC (or clang-cl) x64. MinGW and x86 are not supported.")
endif()
find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tools/generate_native_contract.py" --check
  WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}" COMMAND_ERROR_IS_FATAL ANY)
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
add_compile_options(/W4 /utf-8 "$<$<COMPILE_LANGUAGE:CXX>:/permissive->")
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(COMMON native/preview.cpp native/cursor.cpp native/wave.cpp native/analysis.cpp native/zwicker.cpp native/lufs_meter.cpp)
add_compile_options(/fp:precise)
add_library(reaper_audition_macro SHARED native/transport.cpp ${COMMON}
  native/windows/bar.cpp native/windows/search.cpp native/windows/system.cpp native/windows/drop.cpp native/windows/wave_label.cpp)
target_compile_definitions(reaper_audition_macro PRIVATE AM_EMBEDDED_BAR)
target_link_libraries(reaper_audition_macro PRIVATE user32 gdi32 comctl32 imm32 shell32 dwrite)
set_target_properties(reaper_audition_macro PROPERTIES PREFIX "")
add_library(reaper_audition_macro_worker SHARED native/windows/worker.cpp native/windows/system.cpp)
target_link_libraries(reaper_audition_macro_worker PRIVATE user32 imm32 comctl32)
set_target_properties(reaper_audition_macro_worker PROPERTIES PREFIX "")
if(BUILD_TESTING)
  add_executable(test_zwicker native/test_zwicker.cpp)
  add_executable(test_lufs_meter native/test_lufs_meter.cpp native/lufs_meter.cpp)
  add_executable(test_analysis native/test_analysis.cpp native/zwicker.cpp native/lufs_meter.cpp)
  foreach(test zwicker lufs_meter analysis)
    target_sources(test_${test} PRIVATE native/windows/test_runtime.cpp)
    target_compile_options(test_${test} PRIVATE /UNDEBUG)
    add_test(NAME ${test} COMMAND test_${test})
    set_tests_properties(${test} PROPERTIES WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}" TIMEOUT 120)
  endforeach()
  add_executable(test_quiet_render native/windows/test_quiet_render.cpp)
  target_compile_options(test_quiet_render PRIVATE /UNDEBUG)
  target_link_libraries(test_quiet_render PRIVATE user32 comctl32 imm32)
  add_test(NAME quiet_render COMMAND test_quiet_render)
  set_tests_properties(quiet_render PROPERTIES TIMEOUT 15)
  add_library(reaper_am_test_host SHARED tests/windows_test_host.cpp)
  set_target_properties(reaper_am_test_host PROPERTIES PREFIX "")
  foreach(test transport preview cursor wave)
    add_executable(test_${test} native/test_${test}.cpp)
    target_sources(test_${test} PRIVATE native/windows/test_runtime.cpp)
    target_compile_options(test_${test} PRIVATE /UNDEBUG)
    target_link_libraries(test_${test} PRIVATE user32)
    add_test(NAME ${test} COMMAND test_${test})
    set_tests_properties(${test} PROPERTIES TIMEOUT 45)
  endforeach()
  target_link_libraries(test_cursor PRIVATE gdi32)
  add_executable(test_wave_label native/windows/test_wave_label.cpp native/windows/test_runtime.cpp)
  target_compile_options(test_wave_label PRIVATE /UNDEBUG)
  target_link_libraries(test_wave_label PRIVATE user32 gdi32 dwrite)
  add_test(NAME wave_label COMMAND test_wave_label)
  set_tests_properties(wave_label PROPERTIES TIMEOUT 45)
  add_executable(test_windows native/windows/test_windows.cpp native/windows/bar.cpp native/windows/search.cpp native/windows/system.cpp)
  target_sources(test_windows PRIVATE native/windows/test_runtime.cpp)
  target_compile_options(test_windows PRIVATE /UNDEBUG)
  target_link_libraries(test_windows PRIVATE user32 gdi32 comctl32 imm32 shell32)
  add_test(NAME windows_controls COMMAND test_windows)
  set_tests_properties(windows_controls PROPERTIES WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}" TIMEOUT 45)
  if(AM_TEST_LUA)
    include(FetchContent)
    FetchContent_Declare(lua URL https://www.lua.org/ftp/lua-5.4.8.tar.gz
      URL_HASH SHA256=4f18ddae154e793e46eeab727c59ef1c0c0c2b744e7b94219710d76f530629ae)
    FetchContent_MakeAvailable(lua)
    file(GLOB LUA_SOURCES "${lua_SOURCE_DIR}/src/*.c")
    list(FILTER LUA_SOURCES EXCLUDE REGEX "/luac?\\.c$")
    add_executable(test_lua ${LUA_SOURCES} native/windows/test_lua.cpp native/windows/system.cpp native/windows/test_runtime.cpp)
    target_include_directories(test_lua PRIVATE "${lua_SOURCE_DIR}/src")
    target_link_libraries(test_lua PRIVATE user32 comctl32 imm32)
    add_test(NAME lua_regressions COMMAND test_lua tests/run.lua)
    set_tests_properties(lua_regressions PROPERTIES WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}" TIMEOUT 45)
  endif()
endif()
install(TARGETS reaper_audition_macro reaper_audition_macro_worker RUNTIME DESTINATION build)
